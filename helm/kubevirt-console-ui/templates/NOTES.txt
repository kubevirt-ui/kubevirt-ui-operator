{{- if not .Values.console.registerPlugin }}

Enable the plugin on the cluster console (if not already), preserving other plugins:
  PLUGIN='{{ .Values.consolePlugin.name }}'
  CURRENT="$(oc get consoles.operator.openshift.io cluster -o jsonpath='{.spec.plugins[*]}' 2>/dev/null || true)"
  if echo " ${CURRENT} " | grep -q " ${PLUGIN} "; then
    echo "Console plugin ${PLUGIN} already enabled"
  elif [ -z "$(oc get consoles.operator.openshift.io cluster -o jsonpath='{.spec.plugins}' 2>/dev/null || true)" ]; then
    oc patch consoles.operator.openshift.io cluster --type=merge \
      -p="{\"spec\":{\"plugins\":[\"${PLUGIN}\"]}}"
  else
    oc patch consoles.operator.openshift.io cluster --type=json \
      -p="[{\"op\": \"add\", \"path\": \"/spec/plugins/-\", \"value\": \"${PLUGIN}\"}]"
  fi
{{- end }}
